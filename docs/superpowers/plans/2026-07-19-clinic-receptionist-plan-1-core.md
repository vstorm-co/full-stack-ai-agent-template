# Clinic Receptionist — Plan 1: Core (Generate + Data Layer + Agent + FAQ KB)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A working AI receptionist on web chat: checks availability, requests appointments (with payment estimates), takes staff messages with urgency triage, and answers office FAQs from RAG.

**Architecture:** Generate a project from this template (`fastapi-fullstack`), then customize the generated project at `~/Scripts/Python/clinic_receptionist`. Follow its layered pattern strictly: Routes → Services → Repositories; stateless repo functions; class-based services raising domain exceptions; PydanticAI tools call services through a short-lived DB session.

**Tech Stack:** FastAPI, PostgreSQL (asyncpg), SQLAlchemy 2 + Alembic, PydanticAI, RAG (Qdrant default), ARQ, Next.js 15 (already generated — untouched in this plan).

**Spec:** `docs/superpowers/specs/2026-07-19-clinic-receptionist-design.md` (in the template repo).

**Plans 2 & 3 (separate documents, written after this plan ships):** Plan 2 = staff admin (confirm/decline, availability mgmt, price list, reconciliation report + CSV). Plan 3 = WhatsApp channel, weekly schedule cron, hardening/E2E.

## Global Constraints

- All work after Task 0 happens in the **generated project** at `~/Scripts/Python/clinic_receptionist/backend`, not the template repo.
- Repositories use `db.flush()` + `db.refresh()`, **never** `db.commit()`.
- Routes call services only; services call repos; tools call services.
- `datetime.now(UTC)`, never `datetime.utcnow()`. Timestamps timezone-aware.
- Type hints everywhere; modern syntax (`str | None`, `list[X]`).
- Money is stored as integer cents (`*_cents`), never floats.
- The agent must never give medical advice and never discuss PHI; personal matters become `take_message` handoffs (spec: "No PHI in chat").
- Tests: TDD (write failing test first), ≥80% coverage on new code, test names `test_<action>_<scenario>_<expected_result>`.
- Commit after every green task with conventional-commit messages.

# backend/app/db/models/clinic.py
"""Clinic domain models: scheduling, staff messages, payment estimates."""
import enum
import uuid
from datetime import datetime, time
from uuid import UUID

from sqlalchemy import Boolean, DateTime, Enum, ForeignKey, Integer, String, Text, Time
from sqlalchemy.dialects.postgresql import UUID as PgUUID
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.db.base import Base, TimestampMixin  # adjust to match existing models' imports


class AppointmentStatus(enum.Enum):
    REQUESTED = "requested"
    CONFIRMED = "confirmed"
    DECLINED = "declined"
    CANCELLED = "cancelled"


class EstimateStatus(enum.Enum):
    PENDING = "pending"
    RECONCILED = "reconciled"


class MessageCategory(enum.Enum):
    CALLBACK = "callback"
    BILLING = "billing"
    REFILL = "refill"
    GENERAL = "general"


class MessageStatus(enum.Enum):
    NEW = "new"
    HANDLED = "handled"


class VisitType(Base, TimestampMixin):
    __tablename__ = "visit_types"

    id: Mapped[UUID] = mapped_column(PgUUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    name: Mapped[str] = mapped_column(String(100), unique=True, index=True, nullable=False)
    description: Mapped[str | None] = mapped_column(String(500), nullable=True)
    estimated_cost_cents: Mapped[int] = mapped_column(Integer, nullable=False)
    is_active: Mapped[bool] = mapped_column(Boolean, default=True, nullable=False)

    def __repr__(self) -> str:
        return f"<VisitType(id={self.id}, name={self.name})>"


class Availability(Base, TimestampMixin):
    __tablename__ = "availability"

    id: Mapped[UUID] = mapped_column(PgUUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    weekday: Mapped[int] = mapped_column(Integer, nullable=False)  # 0=Monday … 6=Sunday
    start_time: Mapped[time] = mapped_column(Time, nullable=False)
    end_time: Mapped[time] = mapped_column(Time, nullable=False)
    slot_minutes: Mapped[int] = mapped_column(Integer, default=30, nullable=False)
    is_active: Mapped[bool] = mapped_column(Boolean, default=True, nullable=False)

    def __repr__(self) -> str:
        return f"<Availability(weekday={self.weekday}, {self.start_time}-{self.end_time})>"


class Appointment(Base, TimestampMixin):
    __tablename__ = "appointments"

    id: Mapped[UUID] = mapped_column(PgUUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    patient_name: Mapped[str] = mapped_column(String(255), nullable=False)
    contact: Mapped[str] = mapped_column(String(255), nullable=False)
    reason: Mapped[str | None] = mapped_column(Text, nullable=True)
    visit_type_id: Mapped[UUID] = mapped_column(
        ForeignKey("visit_types.id", ondelete="RESTRICT"), nullable=False
    )
    starts_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), index=True, nullable=False)
    status: Mapped[AppointmentStatus] = mapped_column(
        Enum(AppointmentStatus), default=AppointmentStatus.REQUESTED, nullable=False
    )

    visit_type: Mapped[VisitType] = relationship("VisitType")
    estimate: Mapped["EstimatePayment | None"] = relationship(
        "EstimatePayment", back_populates="appointment", uselist=False, cascade="all, delete-orphan"
    )

    def __repr__(self) -> str:
        return f"<Appointment(id={self.id}, starts_at={self.starts_at}, status={self.status})>"


class EstimatePayment(Base, TimestampMixin):
    __tablename__ = "estimate_payments"

    id: Mapped[UUID] = mapped_column(PgUUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    appointment_id: Mapped[UUID] = mapped_column(
        ForeignKey("appointments.id", ondelete="CASCADE"), unique=True, nullable=False
    )
    visit_type_name: Mapped[str] = mapped_column(String(100), nullable=False)  # snapshot at booking
    estimated_amount_cents: Mapped[int] = mapped_column(Integer, nullable=False)
    status: Mapped[EstimateStatus] = mapped_column(
        Enum(EstimateStatus), default=EstimateStatus.PENDING, nullable=False
    )

    appointment: Mapped[Appointment] = relationship("Appointment", back_populates="estimate")

    def __repr__(self) -> str:
        return f"<EstimatePayment(appointment_id={self.appointment_id}, amount={self.estimated_amount_cents})>"


class StaffMessage(Base, TimestampMixin):
    __tablename__ = "staff_messages"

    id: Mapped[UUID] = mapped_column(PgUUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    patient_name: Mapped[str] = mapped_column(String(255), nullable=False)
    contact: Mapped[str] = mapped_column(String(255), nullable=False)
    category: Mapped[MessageCategory] = mapped_column(
        Enum(MessageCategory), default=MessageCategory.GENERAL, nullable=False
    )
    body: Mapped[str] = mapped_column(Text, nullable=False)
    is_urgent: Mapped[bool] = mapped_column(Boolean, default=False, index=True, nullable=False)
    status: Mapped[MessageStatus] = mapped_column(
        Enum(MessageStatus), default=MessageStatus.NEW, nullable=False
    )

    def __repr__(self) -> str:
        return f"<StaffMessage(id={self.id}, category={self.category}, urgent={self.is_urgent})>"
```

# backend/app/repositories/clinic.py
"""Data access for clinic scheduling, messages, and estimates."""
from datetime import datetime, time
from uuid import UUID

from sqlalchemy import func, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.db.models.clinic import (
    Appointment,
    AppointmentStatus,
    Availability,
    EstimatePayment,
    MessageCategory,
    StaffMessage,
    VisitType,
)


async def create_visit_type(
    db: AsyncSession, *, name: str, estimated_cost_cents: int, description: str | None = None
) -> VisitType:
    vt = VisitType(name=name, estimated_cost_cents=estimated_cost_cents, description=description)
    db.add(vt)
    await db.flush()
    await db.refresh(vt)
    return vt


async def list_active_visit_types(db: AsyncSession) -> list[VisitType]:
    result = await db.execute(
        select(VisitType).where(VisitType.is_active.is_(True)).order_by(VisitType.name)
    )
    return list(result.scalars().all())


async def get_visit_type_by_name(db: AsyncSession, name: str) -> VisitType | None:
    result = await db.execute(
        select(VisitType).where(func.lower(VisitType.name) == name.lower())
    )
    return result.scalar_one_or_none()


async def create_availability(
    db: AsyncSession, *, weekday: int, start_time: time, end_time: time, slot_minutes: int = 30
) -> Availability:
    row = Availability(
        weekday=weekday, start_time=start_time, end_time=end_time, slot_minutes=slot_minutes
    )
    db.add(row)
    await db.flush()
    await db.refresh(row)
    return row


async def list_active_availability(db: AsyncSession) -> list[Availability]:
    result = await db.execute(
        select(Availability)
        .where(Availability.is_active.is_(True))
        .order_by(Availability.weekday, Availability.start_time)
    )
    return list(result.scalars().all())


async def create_appointment(
    db: AsyncSession,
    *,
    patient_name: str,
    contact: str,
    visit_type_id: UUID,
    starts_at: datetime,
    reason: str | None = None,
) -> Appointment:
    appt = Appointment(
        patient_name=patient_name,
        contact=contact,
        visit_type_id=visit_type_id,
        starts_at=starts_at,
        reason=reason,
    )
    db.add(appt)
    await db.flush()
    await db.refresh(appt)
    return appt


async def list_appointments_between(
    db: AsyncSession,
    *,
    start: datetime,
    end: datetime,
    statuses: list[AppointmentStatus] | None = None,
) -> list[Appointment]:
    query = (
        select(Appointment)
        .where(Appointment.starts_at >= start, Appointment.starts_at < end)
        .order_by(Appointment.starts_at)
    )
    if statuses:
        query = query.where(Appointment.status.in_(statuses))
    result = await db.execute(query)
    return list(result.scalars().all())


async def create_estimate(
    db: AsyncSession, *, appointment_id: UUID, visit_type_name: str, estimated_amount_cents: int
) -> EstimatePayment:
    est = EstimatePayment(
        appointment_id=appointment_id,
        visit_type_name=visit_type_name,
        estimated_amount_cents=estimated_amount_cents,
    )
    db.add(est)
    await db.flush()
    await db.refresh(est)
    return est


async def create_staff_message(
    db: AsyncSession,
    *,
    patient_name: str,
    contact: str,
    category: MessageCategory,
    body: str,
    is_urgent: bool = False,
) -> StaffMessage:
    msg = StaffMessage(
        patient_name=patient_name, contact=contact, category=category, body=body, is_urgent=is_urgent
    )
    db.add(msg)
    await db.flush()
    await db.refresh(msg)
    return msg
```

# backend/app/services/clinic.py
"""Business logic for the clinic receptionist: slots, bookings, messages."""
from datetime import UTC, datetime, timedelta

from sqlalchemy.ext.asyncio import AsyncSession

from app.core.exceptions import AlreadyExistsError, NotFoundError
from app.db.models.clinic import (
    Appointment,
    AppointmentStatus,
    MessageCategory,
    StaffMessage,
    VisitType,
)
from app.repositories import clinic as clinic_repo

_BLOCKING = [AppointmentStatus.REQUESTED, AppointmentStatus.CONFIRMED]


class ClinicService:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def open_slots(self, day_start: datetime, day_end: datetime) -> list[datetime]:
        availability = await clinic_repo.list_active_availability(self.db)
        booked = await clinic_repo.list_appointments_between(
            self.db, start=day_start, end=day_end, statuses=_BLOCKING
        )
        booked_starts = {a.starts_at for a in booked}

        slots: list[datetime] = []
        day = day_start
        while day < day_end:
            for window in availability:
                if window.weekday != day.weekday():
                    continue
                cursor = day.replace(
                    hour=window.start_time.hour, minute=window.start_time.minute,
                    second=0, microsecond=0,
                )
                end = day.replace(hour=window.end_time.hour, minute=window.end_time.minute)
                step = timedelta(minutes=window.slot_minutes)
                while cursor + step <= end:
                    if cursor not in booked_starts and day_start <= cursor < day_end:
                        slots.append(cursor)
                    cursor += step
            day += timedelta(days=1)
        return sorted(slots)

    async def request_appointment(
        self,
        *,
        patient_name: str,
        contact: str,
        visit_type_name: str,
        starts_at: datetime,
        reason: str | None = None,
    ) -> Appointment:
        visit_type = await clinic_repo.get_visit_type_by_name(self.db, visit_type_name)
        if visit_type is None or not visit_type.is_active:
            raise NotFoundError(
                message="Unknown visit type", details={"visit_type": visit_type_name}
            )

        window = timedelta(days=1)
        slots = await self.open_slots(
            starts_at.replace(hour=0, minute=0, second=0, microsecond=0),
            starts_at.replace(hour=0, minute=0, second=0, microsecond=0) + window,
        )
        if starts_at not in slots:
            raise AlreadyExistsError(
                message="That time is not available",
                details={"starts_at": starts_at.isoformat()},
            )

        appt = await clinic_repo.create_appointment(
            self.db,
            patient_name=patient_name,
            contact=contact,
            visit_type_id=visit_type.id,
            starts_at=starts_at,
            reason=reason,
        )
        await clinic_repo.create_estimate(
            self.db,
            appointment_id=appt.id,
            visit_type_name=visit_type.name,
            estimated_amount_cents=visit_type.estimated_cost_cents,
        )
        await self.db.refresh(appt, attribute_names=["estimate"])
        return appt

    async def take_message(
        self,
        *,
        patient_name: str,
        contact: str,
        category: str,
        body: str,
        is_urgent: bool = False,
    ) -> StaffMessage:
        try:
            cat = MessageCategory(category.lower().strip())
        except ValueError:
            cat = MessageCategory.GENERAL
        return await clinic_repo.create_staff_message(
            self.db,
            patient_name=patient_name,
            contact=contact,
            category=cat,
            body=body,
            is_urgent=is_urgent,
        )

    async def visit_type_estimates(self) -> list[VisitType]:
        return await clinic_repo.list_active_visit_types(self.db)
```

# backend/app/services/triage.py
"""Urgency detection for incoming patient messages.

Deliberately simple and recall-biased: false positives cost a redundant
"call 911" notice; false negatives cost far more. The agent's system prompt
is the second layer — this keyword scan is the deterministic floor.
"""
import re

_URGENT_PATTERNS = [
    r"chest pain",
    r"can'?t breathe",
    r"trouble breathing",
    r"short(ness)? of breath",
    r"stroke",
    r"heart attack",
    r"unconscious",
    r"not breathing",
    r"seizure",
    r"severe bleeding",
    r"uncontrolled bleeding",
    r"bleeding (a lot|heavily|won'?t stop)",
    r"overdose",
    r"suicid",
    r"end my life",
    r"kill (myself|himself|herself|themselves)",
    r"anaphyla",
    r"severe allergic",
    r"emergency",
]

_URGENT_RE = re.compile("|".join(f"(?:{p})" for p in _URGENT_PATTERNS), re.IGNORECASE)

EMERGENCY_RESPONSE = (
    "This sounds like it could be a medical emergency. Please call 911 now, "
    "or go to the nearest emergency room. If you are in crisis, you can also "
    "call or text 988 (Suicide & Crisis Lifeline). I've flagged your message "
    "for our staff as urgent, but do not wait for a call back."
)


def is_urgent(text: str) -> bool:
    return bool(_URGENT_RE.search(text or ""))

# backend/app/agents/tools/clinic_tools.py
"""Receptionist tools: availability, appointment requests, staff messages.

Each tool opens its own DB session (tools run inside the agent's event loop,
outside any request-scoped session). ``session_factory`` is injectable for tests.
"""
from collections.abc import Callable
from datetime import UTC, datetime, timedelta
from typing import Any

from app.db.session import async_session_maker  # ← substitute the name recorded in Task 0
from app.services.clinic import ClinicService
from app.services.triage import EMERGENCY_RESPONSE, is_urgent

SessionFactory = Callable[[], Any]


def _default_factory() -> Any:
    return async_session_maker()


def _fmt_money(cents: int) -> str:
    return f"${cents / 100:.2f}"


async def tool_check_availability(
    date_iso: str, *, session_factory: SessionFactory = _default_factory
) -> str:
    try:
        day = datetime.fromisoformat(date_iso).replace(tzinfo=UTC, hour=0, minute=0, second=0, microsecond=0)
    except ValueError:
        return f"Could not parse date '{date_iso}'. Please use YYYY-MM-DD."
    async with session_factory() as db:
        slots = await ClinicService(db).open_slots(day, day + timedelta(days=1))
    if not slots:
        return f"No open slots on {day.date().isoformat()}."
    times = ", ".join(s.strftime("%H:%M") for s in slots)
    return f"Open slots on {day.date().isoformat()}: {times}"


async def tool_request_appointment(
    patient_name: str,
    contact: str,
    visit_type: str,
    starts_at_iso: str,
    reason: str | None = None,
    *,
    session_factory: SessionFactory = _default_factory,
) -> str:
    try:
        starts_at = datetime.fromisoformat(starts_at_iso)
        if starts_at.tzinfo is None:
            starts_at = starts_at.replace(tzinfo=UTC)
    except ValueError:
        return f"Could not parse datetime '{starts_at_iso}'. Use ISO format."
    async with session_factory() as db:
        appt = await ClinicService(db).request_appointment(
            patient_name=patient_name,
            contact=contact,
            visit_type_name=visit_type,
            starts_at=starts_at,
            reason=reason,
        )
        estimate = _fmt_money(appt.estimate.estimated_amount_cents) if appt.estimate else "unavailable"
        await db.commit()  # tools own their session lifecycle (no request wrapper)
    return (
        f"Appointment request submitted for {patient_name} — {visit_type} on "
        f"{starts_at.strftime('%A %Y-%m-%d at %H:%M')}. Non-binding cost estimate: {estimate}. "
        "Office staff will confirm shortly."
    )


async def tool_take_message(
    patient_name: str,
    contact: str,
    category: str,
    body: str,
    *,
    session_factory: SessionFactory = _default_factory,
) -> str:
    urgent = is_urgent(body)
    async with session_factory() as db:
        await ClinicService(db).take_message(
            patient_name=patient_name, contact=contact, category=category,
            body=body, is_urgent=urgent,
        )
        await db.commit()
    if urgent:
        return f"URGENT message filed for staff. Relay this to the patient now: {EMERGENCY_RESPONSE}"
    return "Message saved for office staff. They will follow up using the contact provided."
```
